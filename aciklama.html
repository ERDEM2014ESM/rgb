Arduino RGB Led Kontrol Kodu

Aşağıdaki kodu kopyalayarak Arduino IDE uygulamanıza yapıştırabilir ve kartınıza yükleyebilirsiniz.

int brightness1 = 128; 
int brightness2 = 128; 
int brightness3 = 128; 

void setup() {
  Serial.begin(115200);
  pinMode(9, OUTPUT);  // PWM pin
  pinMode(10, OUTPUT); // PWM pin
  pinMode(11, OUTPUT); // PWM pin

  pinMode(7, OUTPUT);
  digitalWrite(7, LOW);

  Serial.println("LED Parlaklik Kontrolu");
  Serial.println("Format: B,G,R (orn: 255,128,64)");
  Serial.println("Komutlar:");
  Serial.println("  '255,255,255' - Tumunu %100 parlaklik");
  Serial.println("  '0,0,0' - Tumunu kapat");
  Serial.println("  '255,0,0' - Sadece ilk LED");
  Serial.println("  '0,255,0' - Sadece ikinci LED");
  Serial.println("  '0,0,255' - Sadece ucuncu LED");
}

void loop() {
  // Seri porttan yeni parlaklık değerleri gelip gelmediğini kontrol et
  if (Serial.available() > 0) {
    // Gelen string'i oku
    String input = Serial.readStringUntil('\n');

    // Virgülle ayrılmış değerleri parse et
    int commaIndex1 = input.indexOf(',');
    int commaIndex2 = input.lastIndexOf(',');

    if (commaIndex1 != -1 && commaIndex2 != -1 && commaIndex1 != commaIndex2) {
      brightness1 = input.substring(0, commaIndex1).toInt();
      brightness2 = input.substring(commaIndex1 + 1, commaIndex2).toInt();
      brightness3 = input.substring(commaIndex2 + 1).toInt();

      // Değerleri 0-255 aralığına sınırla
      brightness1 = constrain(brightness1, 0, 255);
      brightness2 = constrain(brightness2, 0, 255);
      brightness3 = constrain(brightness3, 0, 255);

      Serial.print("Yeni parlaklik degerleri: ");
      Serial.print(brightness1);
      Serial.print(", ");
      Serial.print(brightness2);
      Serial.print(", ");
      Serial.println(brightness3);
    } else {
      Serial.println("Gecersiz format! Lutfen 'B,G,R' formatinda girin.");
    }
  }

  // LED'leri PWM ile kontrol et
  analogWrite(9, brightness1);   // 9 numaralı pine bağlı LED
  analogWrite(10, brightness2);  // 10 numaralı pine bağlı LED
  analogWrite(11, brightness3);  // 11 numaralı pine bağlı LED
}
